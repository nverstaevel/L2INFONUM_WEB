Filter rows and write a new CSV

kept = []
for row in rows:                      # rows: list of dictionaries
    if row["pages"] == "":            # missing value
        continue
    kept.append({"title": row["title"], "pages": int(row["pages"])})

with open("clean.csv", "w", newline="", encoding="utf-8") as f:
    writer = csv.DictWriter(f, fieldnames=["title", "pages"])
    writer.writeheader()
    writer.writerows(kept)

A button, a message, a new chart

let chart = null;
function draw(rows) {
  if (rows.error) {
    document.getElementById("message").textContent = rows.error;
    return;
  }
  if (chart) { chart.destroy(); }
  chart = new Chart(canvas, settings(rows));
}
button.addEventListener("click", load);  // load: fetch, then draw
